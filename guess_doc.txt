Inside rtl/efpga/fabric change the LHQD1 from latch to flip-flop as below:

module LHQD1 (input D, E, output reg Q, QN);
    // gf180mcu has NO latch standard cells, so the original level-sensitive
    // config latch cannot map. The bitstream is written once at reset then held
    // static, so implement each config bit as an edge-triggered DFF that captures
    // on the rising edge of its frame strobe E (FrameStrobe is driven
    // synchronously by the config FSM, so it is glitch-free in practice). This is
    // interface-preserving (no CLK port), so the whole fabric/assembly and the
    // chip-level instantiation are unchanged, and it maps cleanly to gf180 DFFs.
    always @(posedge E)
    begin
        Q  <= D;
        QN <= ~D;
    end
endmodule

OpenROAD found 85 setup violations with a huge worst slack (WNS −150 ns) and tried to repair timing (insert/resize buffers in a loop) — and that loop ran out of memory and aborted (Signal 6 … out of memory).




The root cause (the lesson): a config-static FPGA fabric has no meaningful timing — its config DFFs are clocked by write-once strobes and its switch matrix is full of combinational loops. So every timing-driven part of the flow misbehaves on it. That single fact causes three symptoms: the post-CTS resizer OOMs (what you hit), and the setup / max-slew / max-cap checkers fail at the end. Fix = don't run timing optimization, and don't gate on timing checks.

Now we need to create eFPGA_top macro abstract. Since we can't detailed-route the real fabric on 7.4 GB, we represent eFPGA_top as a sized hard-macro — three views:

.lef = its footprint + every pin on the boundary, with the interior marked as a routing blockage (OBS). The chip router needs only this to place it and route to its pins.
.gds = a placeholder box (the real fabric layout would go here on a big machine).
.bb.v = a (* blackbox *) module so synthesis knows its ports without its innards.

Make the LEF/black-box generator — create work/ip/efpga/make_efpga_macro.py:

'''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''
#!/usr/bin/env python3
# Build a sized ABSTRACT macro for eFPGA_top: LEF (footprint + boundary pins,
# grid-snapped, interior obstructed) + a black-box Verilog.
import re, sys, os
SRC, OUT = sys.argv[1], sys.argv[2]
W, H = 2716.69, 6144.52                 # fabric footprint (from the tile-grid stitch)
os.makedirs(OUT, exist_ok=True)
hdr = open(SRC).read(); hdr = hdr[hdr.find("module eFPGA_top"):]; hdr = hdr[:hdr.find(");")]
ports, pins = [], []
for line in hdr.splitlines():
    m = re.match(r'\s*(input|output)\s*(\[\d+:0\])?\s*([A-Za-z_]\w*)\s*,?', line)
    if not m: continue
    d, w, base = m.group(1), m.group(2) or "", m.group(3)
    ports.append((d, w, base))
    rng = range(int(w[1:-3])+1) if w else [None]      # bus width from [N:0]
    for i in rng: pins.append((f"{base}[{i}]" if i is not None else base, d.upper()))
# black-box Verilog
with open(f"{OUT}/eFPGA_top.bb.v","w") as f:
    f.write("// Black-box of the eFPGA fabric for synthesis (real layout = the macro LEF/GDS).\n")
    f.write("(* blackbox *)\nmodule eFPGA_top (\n")
    f.write(",\n".join(f"  {d} {w}{n}".replace("  "," ").rstrip() for d,w,n in ports))
    f.write("\n);\nendmodule\n")
# LEF with grid-snapped boundary pins
snap = lambda v: round(round(v/0.01)*0.01, 2)
W2,H2 = snap(W),snap(H); per=2*(W2+H2); step=per/len(pins); PW=0.60
def pos(t):
    t%=per
    if t<W2: return (t,0.0)
    t-=W2
    if t<H2: return (W2,t)
    t-=H2
    if t<W2: return (W2-t,H2)
    return (0.0,t-W2)
L=["VERSION 5.8 ;",'BUSBITCHARS "[]" ;',"MACRO eFPGA_top","  CLASS BLOCK ;","  ORIGIN 0 0 ;",
   f"  SIZE {W2:.2f} BY {H2:.2f} ;","  SYMMETRY X Y R90 ;"]
for i,(name,d) in enumerate(pins):
    x,y=pos((i+0.5)*step); x=snap(min(max(x,1.0),W2-1.0)); y=snap(min(max(y,1.0),H2-1.0))
    a,b,c,e=snap(x-PW/2),snap(y-PW/2),snap(x+PW/2),snap(y+PW/2)
    L+=[f"  PIN {name}",f"    DIRECTION {d} ;","    USE SIGNAL ;","    PORT","      LAYER Metal3 ;",
        f"      RECT {a:.2f} {b:.2f} {c:.2f} {e:.2f} ;","    END",f"  END {name}"]
for lay in ["Metal1","Metal2"]:
    L+=["  OBS",f"    LAYER {lay} ;",f"    RECT 2.00 2.00 {snap(W2-2):.2f} {snap(H2-2):.2f} ;","  END"]
L+=["END eFPGA_top","END LIBRARY"]
open(f"{OUT}/eFPGA_top.lef","w").write("\n".join(L)+"\n")
print(f"pins={len(pins)} SIZE={W2}x{H2}")

Inside work/src/ , create chip_mem.sv — unified on-chip memory (8× SRAM) with an OBI arbiter and a loader backdoor:

'''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''
module chip_mem #(
    parameter int unsigned BANKS = 2,
    parameter int unsigned WORDS = BANKS * 512,
    parameter int unsigned AW    = $clog2(WORDS)
) (
    input  logic clk_i, rst_ni,
    input  logic instr_req_i, output logic instr_gnt_o, instr_rvalid_o,
    input  logic [31:0] instr_addr_i, output logic [31:0] instr_rdata_o,
    input  logic data_req_i, output logic data_gnt_o, data_rvalid_o,
    input  logic [31:0] data_addr_i, input logic [3:0] data_be_i,
    input  logic data_we_i, input logic [31:0] data_wdata_i, output logic [31:0] data_rdata_o,
    input  logic load_en_i, input logic [AW-1:0] load_addr_i, input logic [31:0] load_wdata_i
);
  logic sel_data, acc, we; logic [AW-1:0] word_addr; logic [3:0] be; logic [31:0] wdata;
  assign sel_data    = data_req_i;
  assign acc         = (instr_req_i | data_req_i) & ~load_en_i;
  assign data_gnt_o  = data_req_i  & ~load_en_i;
  assign instr_gnt_o = instr_req_i & ~data_req_i & ~load_en_i;
  always_comb begin
    if (load_en_i)      begin word_addr=load_addr_i;          we=1'b1;       be=4'hF;       wdata=load_wdata_i; end
    else if (sel_data)  begin word_addr=data_addr_i[AW+1:2];  we=data_we_i;  be=data_be_i;  wdata=data_wdata_i; end
    else                begin word_addr=instr_addr_i[AW+1:2]; we=1'b0;       be=4'h0;       wdata=32'b0; end
  end
  always_ff @(posedge clk_i or negedge rst_ni)
    if (!rst_ni) begin instr_rvalid_o<=1'b0; data_rvalid_o<=1'b0; end
    else begin instr_rvalid_o<=instr_gnt_o; data_rvalid_o<=data_gnt_o & ~data_we_i; end
  logic [BANKS-1:0] bank_ce; logic [8:0] bank_a; assign bank_a = word_addr[8:0];
  logic [31:0] bank_q [BANKS]; logic [$clog2(BANKS>1?BANKS:2)-1:0] sel_bank_q;
  generate if (BANKS==1) begin : g_one assign bank_ce[0]=1'b1; end
    else begin : g_multi
      logic [$clog2(BANKS)-1:0] sel_bank; assign sel_bank = word_addr[9 +: $clog2(BANKS)];
      always_comb begin bank_ce='0; bank_ce[sel_bank]=1'b1; end
      always_ff @(posedge clk_i) sel_bank_q <= sel_bank;
    end endgenerate
  genvar b,l;
  generate for (b=0;b<BANKS;b++) begin : g_bank
    for (l=0;l<4;l++) begin : g_lane
      logic cen,gwen; logic [7:0] wen;
      assign cen=~(acc & bank_ce[b]); assign gwen=~(we & be[l]); assign wen={8{~(we & be[l])}};
      gf180mcu_fd_ip_sram__sram512x8m8wm1 u_sram (.CLK(clk_i),.CEN(cen),.GWEN(gwen),.WEN(wen),
        .A(bank_a),.D(wdata[l*8 +: 8]),.Q(bank_q[b][l*8 +: 8]));
    end end endgenerate
  logic [31:0] rdata; generate if (BANKS==1) assign rdata=bank_q[0]; else assign rdata=bank_q[sel_bank_q]; endgenerate
  assign instr_rdata_o=rdata; assign data_rdata_o=rdata;
endmodule

Inside work/src/ , create spi_loader.sv — the SPI slave program loader:

''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''

module spi_loader #(parameter int unsigned AW = 10) (
    input  logic clk_i, rst_ni, sclk_i, cs_n_i, mosi_i, output logic miso_o,
    output logic load_en_o, output logic [AW-1:0] load_addr_o, output logic [31:0] load_wdata_o,
    output logic load_done_o
);
  logic [2:0] sclk_s, cs_s, mosi_s;
  always_ff @(posedge clk_i or negedge rst_ni)
    if (!rst_ni) begin sclk_s<='0; cs_s<=3'b111; mosi_s<='0; end
    else begin sclk_s<={sclk_s[1:0],sclk_i}; cs_s<={cs_s[1:0],cs_n_i}; mosi_s<={mosi_s[1:0],mosi_i}; end
  wire sclk_rise=(sclk_s[2:1]==2'b01), cs_active=~cs_s[2], cs_deassert=(cs_s[2:1]==2'b01);
  logic [31:0] shreg; logic [4:0] bitcnt; logic [AW-1:0] waddr;
  assign load_wdata_o=shreg; assign load_addr_o=waddr; assign miso_o=1'b0;
  always_ff @(posedge clk_i or negedge rst_ni)
    if (!rst_ni) begin shreg<=32'b0; bitcnt<=5'd0; waddr<='0; load_en_o<=1'b0; load_done_o<=1'b0; end
    else begin
      load_en_o<=1'b0;
      if (cs_active && sclk_rise) begin
        shreg<={shreg[30:0],mosi_s[2]}; bitcnt<=bitcnt+5'd1;
        if (bitcnt==5'd31) begin load_en_o<=1'b1; bitcnt<=5'd0; end
      end
      if (load_en_o) waddr<=waddr+1'b1;
      if (cs_deassert) load_done_o<=1'b1;
    end
endmodule

''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''''


chip_mem: 8 SRAM macros = two 512×32 words (4 KB). A priority arbiter gives the data port precedence and stalls the instruction port on a same-cycle clash (by withholding instr_gnt). The OBI timing is "grant now, data next cycle." The load_* backdoor lets the loader write memory while the CPU is held in reset.
spi_loader: samples the async SPI pins in the core clock domain (2-flop synchronizers), shifts in 32-bit words MSB-first, writes each to the backdoor at an auto-incrementing address, and raises load_done when chip-select deasserts — which we'll use to release the CPU from reset. This is the "load the program over SPI instead of hardcoding it" mechanism.

// Full SoC user-logic: CPU+eFPGA (cv32e40x_top) + on-chip memory (chip_mem) +
// SPI program loader (spi_loader). The CPU is held in reset until the loader
// has streamed the program into SRAM (load_done).
module chip_core #(
    parameter int unsigned MEM_BANKS = 2,
    parameter int unsigned MEM_AW    = $clog2(MEM_BANKS*512)
) (
    input  logic clk_i, rst_ni, fetch_enable_i,
    input  logic [31:0] boot_addr_i,
    input  logic spi_sclk_i, spi_cs_n_i, spi_mosi_i, output logic spi_miso_o,
    output logic core_sleep_o
);
  logic load_en, load_done; logic [MEM_AW-1:0] load_addr; logic [31:0] load_wdata;
  spi_loader #(.AW(MEM_AW)) u_loader (
      .clk_i(clk_i), .rst_ni(rst_ni),
      .sclk_i(spi_sclk_i), .cs_n_i(spi_cs_n_i), .mosi_i(spi_mosi_i), .miso_o(spi_miso_o),
      .load_en_o(load_en), .load_addr_o(load_addr), .load_wdata_o(load_wdata), .load_done_o(load_done));
  logic core_rst_n; assign core_rst_n = rst_ni & load_done;   // boot CPU after load
  logic instr_req, instr_gnt, instr_rvalid; logic [31:0] instr_addr, instr_rdata;
  logic data_req, data_gnt, data_rvalid, data_we; logic [31:0] data_addr, data_wdata, data_rdata; logic [3:0] data_be;
  cv32e40x_top u_cpu (
      .clk_i(clk_i), .rst_ni(core_rst_n), .fetch_enable_i(fetch_enable_i),
      .boot_addr_i(boot_addr_i), .dm_halt_addr_i(32'h0), .mhartid_i(32'h0),
      .instr_req_o(instr_req), .instr_gnt_i(instr_gnt), .instr_rvalid_i(instr_rvalid),
      .instr_addr_o(instr_addr), .instr_rdata_i(instr_rdata),
      .data_req_o(data_req), .data_gnt_i(data_gnt), .data_rvalid_i(data_rvalid),
      .data_addr_o(data_addr), .data_be_o(data_be), .data_we_o(data_we),
      .data_wdata_o(data_wdata), .data_rdata_i(data_rdata), .core_sleep_o(core_sleep_o));
  chip_mem #(.BANKS(MEM_BANKS)) u_mem (
      .clk_i(clk_i), .rst_ni(rst_ni),
      .instr_req_i(instr_req), .instr_gnt_o(instr_gnt), .instr_rvalid_o(instr_rvalid),
      .instr_addr_i(instr_addr), .instr_rdata_o(instr_rdata),
      .data_req_i(data_req), .data_gnt_o(data_gnt), .data_rvalid_o(data_rvalid),
      .data_addr_i(data_addr), .data_be_i(data_be), .data_we_i(data_we),
      .data_wdata_i(data_wdata), .data_rdata_o(data_rdata),
      .load_en_i(load_en), .load_addr_i(load_addr), .load_wdata_i(load_wdata));
endmodule
